【三、第二段·自动断网脚本】🔴 **关网前必须先把下面三件事都写进脚本**：独立兜底、trap 恢复、采样循环。
写 `/tmp/ag/outage.sh` 并后台跑（⛔ 前台跑会连 CC 自己一起卡住）：

```bash
#!/bin/bash
DEV=en1
LOG=/tmp/ag/samples.log
# ① 独立兜底：本脚本即使被 SIGKILL（trap 不执行）也能把网开回来
( sleep 300; networksetup -setairportpower "$DEV" on ) >/dev/null 2>&1 &
disown
# ② trap 恢复：正常退出/中断都恢复
restore() { networksetup -setairportpower "$DEV" on; echo "$(date '+%T') RESTORE-TRAP" >> "$LOG"; }
trap restore EXIT INT TERM
sample() {
  echo "=== $(date '+%T.%N') phase=$1 ===" >> "$LOG"
  cat /Users/paulshao/Projects/HumanResource/data/liaison/liveness.json >> "$LOG" 2>&1; echo >> "$LOG"
  sqlite3 /Users/paulshao/Projects/HumanResource/data/liaison.db \
    "select id,started_at,ended_at from liaison_outage_window;" >> "$LOG" 2>&1
  tail -3 /tmp/ag/stderr.txt >> "$LOG"
}
for i in $(seq 1 8);  do sample BEFORE;  sleep 2; done   # 断网前基线 ~16s
echo "$(date '+%T') >>> WIFI OFF" >> "$LOG"
networksetup -setairportpower "$DEV" off
for i in $(seq 1 45); do sample OUTAGE;  sleep 2; done   # 断网 ~90s
echo "$(date '+%T') >>> WIFI ON" >> "$LOG"
networksetup -setairportpower "$DEV" on
for i in $(seq 1 60); do sample RECOVER; sleep 2; done   # 复网 ~120s
echo "$(date '+%T') >>> DONE" >> "$LOG"
```

**起完就不要再发任何工具调用**，等它跑完（约 4 分钟）——期间 CC 无网。
