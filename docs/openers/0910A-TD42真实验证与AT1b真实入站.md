脚本落 `/tmp/aa/outage.sh`（三件事在关网**前**写死：独立兜底 `sleep 900` 复网、
`trap restore EXIT INT TERM`、每 5 秒采样）：

```
#!/bin/bash
DEV=<实测设备名>; REPO=<仓库根绝对路径>; LOG=/tmp/aa/samples.log; mkdir -p /tmp/aa
( sleep 900; networksetup -setairportpower "$DEV" on ) >/dev/null 2>&1 & disown
restore() { networksetup -setairportpower "$DEV" on; echo "$(date '+%T') RESTORE-TRAP" >> "$LOG"; }
trap restore EXIT INT TERM
sample() { echo "=== $(date '+%T') phase=$1 pid=$(launchctl print gui/$UID/com.zhuopin.hr.liaison 2>/dev/null | awk '/pid =/{print $3}') ===" >> "$LOG";
           cat "$REPO/data/liaison/liveness.json" >> "$LOG" 2>&1; echo >> "$LOG";
           cat "$REPO/data/liaison/watchdog.json" >> "$LOG" 2>&1; echo >> "$LOG";
           grep -E "判定连接假死|看门狗终止进程|连接假死|拉起" "$REPO/data/liaison/logs/liaison.log" | tail -3 >> "$LOG"; }
for i in $(seq 1 3);   do sample BEFORE;  sleep 5; done
echo "$(date '+%T') >>> WIFI OFF" >> "$LOG"; networksetup -setairportpower "$DEV" off
for i in $(seq 1 108); do sample OUTAGE;  sleep 5; done          # 540 秒 = 9 分钟
echo "$(date '+%T') >>> WIFI ON" >> "$LOG";  networksetup -setairportpower "$DEV" on
for i in $(seq 1 36);  do sample RECOVER; sleep 5; done          # 复网后观察 3 分钟
echo "$(date '+%T') >>> DONE" >> "$LOG"
```
